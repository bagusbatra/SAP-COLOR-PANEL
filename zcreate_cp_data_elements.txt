*&---------------------------------------------------------------------*
*& Form CREATE_DOMAIN
*&---------------------------------------------------------------------*
FORM create_domain USING p_domname TYPE clike
                         p_dtype TYPE clike
                         p_length TYPE i
                         p_ddtext TYPE clike.

  DATA: ls_dd01v TYPE dd01v,
        ls_result TYPE ty_result.

  ls_result-rollname = p_domname.

  " Check exists
  DATA: ls_check TYPE dd01v.
  CALL FUNCTION 'DDIF_DOMA_GET'
    EXPORTING
      name     = p_domname
      state    = 'A'
    IMPORTING
      dd01v_wa = ls_check
    EXCEPTIONS OTHERS = 1.

  IF sy-subrc = 0 AND ls_check-domname IS NOT INITIAL.
    ls_result-status  = 'EXISTS'.
    ls_result-message = 'Domain exists'.
    APPEND ls_result TO gt_result.
    RETURN.
  ENDIF.

  IF p_test = 'X'.
    ls_result-status  = 'WOULD CREATE'.
    ls_result-message = 'Test mode'.
    APPEND ls_result TO gt_result.
    RETURN.
  ENDIF.

  CLEAR ls_dd01v.
  ls_dd01v-domname    = p_domname.
  ls_dd01v-ddlanguage = sy-langu.
  ls_dd01v-datatype   = p_dtype.
  ls_dd01v-leng       = p_length.
  ls_dd01v-ddtext     = p_ddtext.

  CALL FUNCTION 'DDIF_DOMA_PUT'
    EXPORTING
      name      = p_domname
      dd01v_wa  = ls_dd01v
    EXCEPTIONS OTHERS = 1.

  IF sy-subrc = 0.
    ls_result-status  = 'CREATED'.
    ls_result-message = 'Domain created'.
  ELSE.
    ls_result-status  = 'ERROR'.
    ls_result-message = 'Domain creation failed'.
  ENDIF.

  APPEND ls_result TO gt_result.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form CREATE_DE
*&---------------------------------------------------------------------*
FORM create_de USING p_rollname TYPE clike
                     p_domname TYPE clike
                     p_datatype TYPE clike
                     p_ddtext TYPE clike.

  DATA: ls_dd04v TYPE dd04v,
        ls_result TYPE ty_result.

  ls_result-rollname = p_rollname.

  " Check exists
  DATA: ls_check TYPE dd04v.
  CALL FUNCTION 'DDIF_DTEL_GET'
    EXPORTING
      name     = p_rollname
      state    = 'A'
    IMPORTING
      dd04v_wa = ls_check
    EXCEPTIONS OTHERS = 1.

  IF sy-subrc = 0 AND ls_check-rollname IS NOT INITIAL.
    ls_result-status  = 'EXISTS'.
    ls_result-message = 'Data Element exists'.
    APPEND ls_result TO gt_result.
    RETURN.
  ENDIF.

  IF p_test = 'X'.
    ls_result-status  = 'WOULD CREATE'.
    ls_result-message = 'Test mode'.
    APPEND ls_result TO gt_result.
    RETURN.
  ENDIF.

  CLEAR ls_dd04v.
  ls_dd04v-rollname    = p_rollname.
  ls_dd04v-ddlanguage  = sy-langu.
  ls_dd04v-domname     = p_domname.
  ls_dd04v-ddtext      = p_ddtext.
  ls_dd04v-scrtext_s   = p_ddtext(10).
  ls_dd04v-scrtext_m   = p_ddtext(20).
  ls_dd04v-scrtext_l   = p_ddtext(40).

  CALL FUNCTION 'DDIF_DTEL_PUT'
    EXPORTING
      name      = p_rollname
      dd04v_wa  = ls_dd04v
    EXCEPTIONS OTHERS = 1.

  IF sy-subrc = 0.
    ls_result-status  = 'CREATED'.
    ls_result-message = 'Data Element created'.
  ELSE.
    ls_result-status  = 'ERROR'.
    ls_result-message = 'DE creation failed'.
  ENDIF.

  APPEND ls_result TO gt_result.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form ACTIVATE_ALL
*&---------------------------------------------------------------------*
FORM activate_all.

  DATA: ls_r TYPE ty_result.

  " Activate domains first
  LOOP AT gt_result INTO ls_r WHERE status = 'CREATED'
    AND rollname CP 'ZCP_D_*'.
    CALL FUNCTION 'DDIF_DOMA_ACTIVATE'
      EXPORTING name = ls_r-rollname
      EXCEPTIONS OTHERS = 1.
    IF sy-subrc = 0.
      ls_r-status = 'ACTIVATED'.
      MODIFY gt_result FROM ls_r.
    ENDIF.
  ENDLOOP.

  " Then activate data elements
  LOOP AT gt_result INTO ls_r WHERE status = 'CREATED'.
    CALL FUNCTION 'DDIF_DTEL_ACTIVATE'
      EXPORTING name = ls_r-rollname
      EXCEPTIONS OTHERS = 1.
    IF sy-subrc = 0.
      ls_r-status = 'ACTIVATED'.
      MODIFY gt_result FROM ls_r.
    ENDIF.
  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form DISPLAY_RESULTS
*&---------------------------------------------------------------------*
FORM display_results.

  WRITE: / '========================================================'.
  WRITE: / 'Color Panel Data Elements Creation Report'.
  WRITE: / 'Date:', sy-datum, 'Time:', sy-uzeit.
  WRITE: / 'Mode:', COND char20( WHEN p_test = 'X' THEN 'TEST' ELSE 'CREATE' ).
  WRITE: / '========================================================'.
  WRITE: /.

  DATA: lv_domain_count TYPE i VALUE 0,
        lv_de_count TYPE i VALUE 0.

  LOOP AT gt_result INTO DATA(ls_r).
    WRITE: / ls_r-rollname(20), '|', ls_r-status(15), '|', ls_r-message.
    IF ls_r-rollname CP 'ZCP_D_*'.
      lv_domain_count = lv_domain_count + 1.
    ELSE.
      lv_de_count = lv_de_count + 1.
    ENDIF.
  ENDLOOP.

  WRITE: /.
  WRITE: / 'Domains:', lv_domain_count, '| Data Elements:', lv_de_count.
  WRITE: /.
  WRITE: / 'NEXT: Run ZCREATE_COLOR_PANEL_TABLES to create tables.'.

ENDFORM.
