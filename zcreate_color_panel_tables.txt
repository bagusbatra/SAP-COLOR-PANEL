*&---------------------------------------------------------------------*
*& Report ZCREATE_COLOR_PANEL_TABLES
*& Create all Z-tables for Color Panel Management System
*& Run in SE38 → Execute → Tables created in DDIC
*& After run: activate all tables via SE11 or mass activate
*&---------------------------------------------------------------------*
*& WARNING: Run in $TMP (local) first. Transport later.
*& Tables: ZCP_USER, ZCP_BUYER, ZCP_WC, ZCP_DCP_HDR, ZCP_DCP_ITEM,
*&         ZCP_MCP, ZCP_CP, ZCP_BORROW, ZCP_BORROW_ITM,
*&         ZCP_BORROW_ARC, ZCP_AUDIT_LOG
*&---------------------------------------------------------------------*
REPORT zcreate_color_panel_tables.

*----------------------------------------------------------------------*
* Selection screen
*----------------------------------------------------------------------*
PARAMETERS: p_test AS CHECKBOX DEFAULT 'X'.  " Test mode (display only)
PARAMETERS: p_act  AS CHECKBOX.              " Activate after create

*----------------------------------------------------------------------*
* Types
*----------------------------------------------------------------------*
TYPES: BEGIN OF ty_result,
         tabname   TYPE dd02l-tabname,
         status    TYPE char20,
         message   TYPE char100,
       END OF ty_result,
       tt_result TYPE STANDARD TABLE OF ty_result WITH DEFAULT KEY.

*----------------------------------------------------------------------*
* Global data
*----------------------------------------------------------------------*
DATA: gt_result TYPE tt_result,
      gv_dd02v  TYPE dd02v,
      gv_dd03p  TYPE TABLE OF dd03p WITH HEADER LINE,
      gv_subrc  TYPE sy-subrc.

*----------------------------------------------------------------------*
* Main
*----------------------------------------------------------------------*
START-OF-SELECTION.

  PERFORM create_table_zcp_user.
  PERFORM create_table_zcp_buyer.
  PERFORM create_table_zcp_wc.
  PERFORM create_table_zcp_dcp_hdr.
  PERFORM create_table_zcp_dcp_item.
  PERFORM create_table_zcp_mcp.
  PERFORM create_table_zcp_cp.
  PERFORM create_table_zcp_borrow.
  PERFORM create_table_zcp_borrow_itm.
  PERFORM create_table_zcp_borrow_arc.
  PERFORM create_table_zcp_audit_log.

* Activate if requested
  IF p_act = 'X'.
    PERFORM activate_all_tables.
  ENDIF.

* Display results
  PERFORM display_results.

  PERFORM add_field USING 'USER_ROLE'   'ZCP_ROLE'      '' '' 'IT/BORROWER/QC'.
  PERFORM add_field USING 'DETAILS'     'ZCP_DETAILS'   '' '' 'Deskripsi Aksi'.
  PERFORM add_field USING 'TIMESTAMP'   'ZCP_TIMESTAMP' '' '' 'Waktu Aksi'.
  PERFORM add_field USING 'IP_ADDR'     'ZCP_IP_ADDR'   '' '' 'IP Address'.

  PERFORM put_table USING 'ZCP_AUDIT_LOG' gv_dd02v gv_dd03p[].

ENDFORM.

*&---------------------------------------------------------------------*
*& Form ADD_FIELD
*&---------------------------------------------------------------------*
FORM add_field USING p_fname TYPE clike
                     p_rollname TYPE clike
                     p_keyflag TYPE clike
                     p_notnull TYPE clike
                     p_ddtext TYPE clike.

  CLEAR gv_dd03p.
  gv_dd03p-fieldname  = p_fname.
  gv_dd03p-rollname   = p_rollname.
  gv_dd03p-keyflag    = p_keyflag.
  gv_dd03p-notnull    = p_notnull.
  gv_dd03p-ddlanguage = sy-langu.
  gv_dd03p-ddtext     = p_ddtext.
  APPEND gv_dd03p.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form PUT_TABLE
*&---------------------------------------------------------------------*
FORM put_table USING p_tabname TYPE clike
                     p_dd02v TYPE dd02v
                     p_dd03p TYPE TABLE.

  DATA: lv_subrc  TYPE sy-subrc,
        lv_msg    TYPE char100,
        ls_result TYPE ty_result.

  ls_result-tabname = p_tabname.

* Check if table already exists
  DATA: ls_dd02v_check TYPE dd02v.
  CALL FUNCTION 'DDIF_TABL_GET'
    EXPORTING
      name          = p_tabname
      state         = 'A'
    IMPORTING
      dd02v_wa      = ls_dd02v_check
    EXCEPTIONS
      illegal_input = 1
      OTHERS        = 2.

  IF sy-subrc = 0 AND ls_dd02v_check-tabname IS NOT INITIAL.
    ls_result-status  = 'EXISTS'.
    ls_result-message = 'Table already exists - skipped'.
    APPEND ls_result TO gt_result.
    RETURN.
  ENDIF.

* Test mode: display only
  IF p_test = 'X'.
    ls_result-status  = 'WOULD CREATE'.
    ls_result-message = 'Test mode - table would be created'.
    APPEND ls_result TO gt_result.
    RETURN.
  ENDIF.

* Create table
  CALL FUNCTION 'DDIF_TABL_PUT'
    EXPORTING
      name      = p_tabname
      dd02v_wa  = p_dd02v
      dd03p_tab = p_dd03p
    EXCEPTIONS
      OTHERS    = 1.

  IF sy-subrc = 0.
    ls_result-status  = 'CREATED'.
    ls_result-message = 'Table created successfully in DDIC'.
  ELSE.
    ls_result-status  = 'ERROR'.
    ls_result-message = 'Failed to create table'.
  ENDIF.

  APPEND ls_result TO gt_result.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form ACTIVATE_ALL_TABLES
*&---------------------------------------------------------------------*
FORM activate_all_tables.

  DATA: lv_tabname TYPE dd02l-tabname.
  DATA: ls_result TYPE ty_result.

  LOOP AT gt_result INTO ls_result WHERE status = 'CREATED'.
    lv_tabname = ls_result-tabname.

    CALL FUNCTION 'DDIF_TABL_ACTIVATE'
      EXPORTING
        name      = lv_tabname
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.

    IF sy-subrc = 0.
      ls_result-status  = 'ACTIVATED'.
      ls_result-message = 'Table activated'.
    ELSE.
      ls_result-status  = 'ACTIV_ERR'.
      ls_result-message = 'Activation failed - activate manually in SE11'.
    ENDIF.

    MODIFY gt_result FROM ls_result.
  ENDLOOP.

ENDFORM.

ENDFORM.
